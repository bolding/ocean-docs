# Bias Correction Configuration — Example
#
# Corrects future atmospheric forcing files from a CMIP6 model so that
# their climatology matches a gridded observational reference.
#
# Workflow
# --------
#   1. Load observational reference  (ref)  for the calibration period.
#   2. Load historical model run     (hist) for the same calibration period.
#   3. Load future model run         (future) for the projection period.
#   4. Regrid everything to the model/output grid if grids differ.
#   5. Apply bias correction per variable, per calendar month.
#   6. Write one NetCDF per output year.
#
# Run with:
#   bc-correct --config config/bc_correct_example.yaml

# ── Metadata ─────────────────────────────────────────────────────────────────
# scenario is substituted for {scenario} in future_model URIs, output.dir, and
# filename patterns; also used by bc-diagnostics to place outputs under
#   <output.analyses_dir>/scenarios/<scenario>/
# model is substituted for {model} throughout all CMIP6 URIs and output.dir.
# Override on the command line: bc correct --config ... --model MPI-ESM1-2-LR
metadata:
  scenario: ssp245
  model:    MPI-ESM1-2-HR

# ── CMIP6 settings ────────────────────────────────────────────────────────────
# access: how CMIP6 data is loaded.
#
#   pangeo     — Pangeo GCS Zarr mirror (default).  Streams only the requested
#                spatial domain and time range via xarray + Dask.  Recommended
#                for atmospheric daily data: a global daily field (e.g. tas/day
#                for MPI-ESM1-2-HR) is ~220 MB/file × 33 files = 7 GB just for
#                historical.  Pangeo transfers only your bbox subset, which is
#                far faster than bulk download.
#
#   esgf       — cascade: Pangeo first, ESGF OPeNDAP second.  Use when Pangeo
#                is missing a variable (e.g. some radiation components).  Falls
#                back automatically.
#
#   esgf_direct — ESGF OPeNDAP only (no Pangeo attempt).  Streams the subset
#                you request; slower per-variable but no local storage needed.
#
#   NOTE: http (bulk download) is NOT recommended for atmospheric daily data —
#   a single scenario × variable can be 4–7 GB; 10 variables × 2 scenarios
#   easily exceeds 100 GB.  Use Pangeo or OPeNDAP streaming instead.
#
# member_defaults: per-model ensemble member IDs.  The entries below mirror the
# built-in defaults that are always applied automatically (all verified via ESGF).
# Add further models here or on the command line with
# --member-defaults MODEL:MEMBER_ID to extend or override.
#
# Both URI forms are handled — {member_id} placeholder and hardcoded segment:
#   uri: "cmip6://CNRM-ESM2-1/historical/tas/day/{member_id}"   # → r1i1p1f2
#   uri: "cmip6://CNRM-ESM2-1/historical/tas/day/r1i1p1f1"     # → overridden to r1i1p1f2
#
# r1i1p1f1 (standard) — no entry needed:
#   MPI-ESM1-2-HR, MPI-ESM1-2-LR, GFDL-ESM4, GFDL-CM4,
#   IPSL-CM6A-LR, EC-Earth3, EC-Earth3-Veg,
#   NorESM2-MM, NorESM2-LM, ACCESS-CM2, ACCESS-ESM1-5,
#   CESM2, CESM2-WACCM, BCC-CSM2-MR, MIROC6, INM-CM5-0, MRI-ESM2-0
cmip6:
  access: pangeo    # pangeo (default) | esgf (Pangeo→OPeNDAP) | esgf_direct (OPeNDAP only)
  member_defaults:
    CNRM-ESM2-1:     r1i1p1f2
    CNRM-CM6-1:      r1i1p1f2
    CNRM-CM6-1-HR:   r1i1p1f2
    UKESM1-0-LL:     r1i1p1f2
    HadGEM3-GC31-LL: r1i1p1f3
    HadGEM3-GC31-MM: r1i1p1f3
  # skip_variables: variables to exclude per model and scenario (e.g. when
  # the required CMIP6 data is unavailable on ESGF).
  # MPI-ESM1-2-HR does not publish rlds/rlus at daily frequency for ssp126.
  skip_variables:
    MPI-ESM1-2-HR:
      ssp126: [net_lw]

# ── Calibration period ───────────────────────────────────────────────────────
# The reference and historical datasets must cover this period.
# `end` must NOT be later than "2014-12-31": CMIP6's `historical`
# experiment is a fixed, protocol-bounded 1850-2014 for every model, no
# exceptions. A later end date doesn't error -- it silently truncates
# the CMIP6 (hist) side of the QDM fit while the reference (ref) side
# still uses the full window, an asymmetric calibration baseline that
# bakes in a spurious "model bias" from whatever real trend exists in
# the extra ref-only years. Real incident: this file previously showed
# "2019-12-31" here, undetected in production for months.
calibration:
  start: "1990-01-01"
  end:   "2014-12-31"

# ── Future period ─────────────────────────────────────────────────────────────
# The corrected output will cover this period.
future:
  start: "2020-01-01"
  end:   "2099-12-31"

# ── Historical output period (optional) ──────────────────────────────────────
# Used by:  bc correct --historical  (or --historical-start/--historical-end)
# Defines the period of bias-corrected CMIP6 historical output.
# Falls back to calibration period if not set.
# Typical use: produce consistent forcing for the full historical run (e.g.
# 1950–2014) even though calibration only uses a subset (e.g. 1990–2019).
historical_period:
  start: "1950-01-01"
  end:   "2014-12-31"

# ── Spatial domain ────────────────────────────────────────────────────────────
# All datasets are subset to this domain before correction.
domain:
  lon_min: -20.0
  lon_max:  30.0
  lat_min:  45.0
  lat_max:  75.0

# ── Variables ─────────────────────────────────────────────────────────────────
# List the variables to correct.  For each variable you must provide:
#   name       — variable name in the model files
#   kind       — '+' (additive, e.g. temperature) or '*' (multiplicative, e.g. precip)
#              — omit to use the built-in default for the variable name
#
variables:
  - name: tas
    kind: "+"
    regridding:
      method: bilinear
  - name: pr
    kind: "*"
    regridding:
      method: conservative   # preserves area-integrated totals for precipitation
  # Net shortwave = rsds − rsus (ERA5 reference: ssr).
  # Correcting the net rather than the downwelling component accounts for
  # differences in surface albedo parameterisations between ERA5 and CMIP6.
  # See guides/radiation-bc.md for background and availability notes.
  - name: net_sw
    kind: "*"             # multiplicative — net SW is always ≥ 0
    regridding:
      method: conservative

  # Net longwave = rlds − rlus (ERA5 reference: str).
  # str is negative (ocean loses LW heat); additive kind handles signed values.
  - name: net_lw
    kind: "+"             # additive — net LW is always ≤ 0
    regridding:
      method: conservative

  # ── Alternative: downwelling-only correction (legacy / CMIP7 downwelling-only) ──
  # If rsus/rlus are unavailable, fall back to correcting the downwelling
  # component directly.  Replace net_sw/net_lw above with the two entries below.
  # - name: rsds
  #   kind: "*"
  #   regridding:
  #     method: conservative
  # - name: rlds
  #   kind: "*"
  #   regridding:
  #     method: conservative
  #
  # ── CMIP7 note: direct net radiation ─────────────────────────────────────────
  # CMIP7 may publish net surface radiation directly (e.g. rsns / rlns).
  # If so, replace the composite blocks below in reference/historical/future_model
  # with simple single-source configs pointing to those variables, keeping the
  # ERA5 ssr/str reference entries unchanged.
  - name: uas
    kind: "+"
    regridding:
      method: bilinear
  - name: vas
    kind: "+"
    regridding:
      method: bilinear
  - name: ps
    kind: "+"    # additive — both ERA5 sp and CMIP6 ps are in Pa
    regridding:
      method: bilinear
  - name: psl
    kind: "+"    # additive — both ERA5 mslp and CMIP6 psl are in Pa
    regridding:
      method: bilinear
  - name: huss
    kind: "*"    # multiplicative — non-negative ratio
    regridding:
      method: bilinear
  - name: evspsbl
    kind: "*"    # multiplicative — non-negative flux after sign correction
    regridding:
      method: bilinear  # spatially smooth; bilinear correct for per-area flux

# ── Correction method ─────────────────────────────────────────────────────────
correction:
  method: qdm          # 'delta' | 'qm' | 'qdm' (recommended for projections)
                       # 'qdm_v1': the kernel called qdm before 2026-11
                       # (effectively qm); only to reproduce old products
  n_quantiles: 100     # number of quantile bins for 'qm' and 'qdm'
  window_years: 30     # qdm: future values are ranked within this moving window
  step_years: 10       # qdm: the window moves in steps of this many years

# ── Regridding ────────────────────────────────────────────────────────────────
# Spatial interpolation method used when regridding model → reference grid.
# Must be set per variable via the 'regridding:' block inside each variable entry.
# Options: bilinear | conservative | nearest_s2d | nearest_d2s | patch
#   bilinear    — fast, smooth; good for most continuous fields
#   conservative — preserves area-integrated totals; best for fluxes and precip
#   nearest_s2d — nearest-neighbour; useful for categorical or masked fields

# ── Observational reference ───────────────────────────────────────────────────
# Any source supported by DataLoader:
#   - local NetCDF path (supports {year} placeholder)
#   - copernicus:// URI  (requires: copernicusmarine login)
#   - cmip6:// URI       (via ESGF; requires: pip install requests; optional: pip install pydap)
#
reference:
  tas:
    source: local
    path:   "${ERA5_FOLDER}/era5_t2m_{year}.nc"
    variable: t2m        # name inside the file (if different from 'tas')
    # ERA5 t2m and CMIP6 tas are both in Kelvin — no conversion needed.
    # Add offset: -273.15 only if one source is in °C and the other in K.
    resample: daily      # the model is daily: correct daily means against
                         # daily means. An hourly reference gives the daily
                         # values the spread of single hours (tas: +6 % in
                         # winter to +14 % in summer, tails 1-2 K, single
                         # cells up to 7 K; NSe domain, ERA5 1990-2014), and
                         # disaggregation then adds the diurnal cycle again.
                         # Also cuts memory from ~70 GB to a few GB.

  pr:
    source: local
    path:   "${ERA5_FOLDER}/era5_tp_{year}.nc"
    variable: tp
    resample: daily_sum  # hourly accumulations summed to daily totals
    # ERA5 tp is stored as an accumulation in metres (m per time step).
    # CMIP6 pr is in kg m-2 s-1 (= mm/s).
    # DO NOT set scale_factor here — _align_units() auto-detects the ERA5
    # m/step unit via the 'units' attribute and converts automatically using
    # the actual time-step length (1 day after daily_sum).  Adding a manual
    # scale_factor would cause a double conversion.

  # ERA5 provides net radiation directly — no composite needed for the reference.
  # ERA5 ssr = surface net solar radiation (ssrd − reflected SW); always ≥ 0.
  # ERA5 str = surface net thermal radiation (strd − surface upwelling LW); always ≤ 0.
  # Both are accumulated J/m²; scale_factor converts to W/m² for hourly files.
  # For daily ERA5 files use scale_factor: 0.000011574 (÷ 86400 s/day) instead.
  net_sw:
    source: local
    path:     "${ERA5_FOLDER}/era5_ssr_{year}.nc"
    variable: ssr
    scale_factor: 0.000277778   # J/m² (hourly accumulation) → W/m²
    resample: daily              # REQUIRED: night-time zeros make hourly distribution
                                 # incompatible with daily CMIP6 net_sw

  net_lw:
    source: local
    path:     "${ERA5_FOLDER}/era5_str_{year}.nc"
    variable: str
    scale_factor: 0.000277778   # same as net_sw
    resample: daily

  uas:
    source: local
    path:   "${ERA5_FOLDER}/era5_u10_{year}.nc"
    variable: u10
    resample: daily      # as tas (hourly reference: spread +9-11 %)

  vas:
    source: local
    path:   "${ERA5_FOLDER}/era5_v10_{year}.nc"
    variable: v10
    resample: daily

  ps:
    source: local
    path:   "${ERA5_FOLDER}/era5_sp_{year}.nc"
    variable: sp
    resample: daily
    # ERA5 sp and CMIP6 ps are both in Pa — no unit conversion needed.

  psl:
    source: local
    path:   "${ERA5_FOLDER}/era5_mslp_{year}.nc"
    variable: mslp
    resample: daily      # (hourly reference: spread +3 %)
    # ERA5 mslp and CMIP6 psl are both in Pa — no unit conversion needed.

  huss:
    source: local
    path:        "${ERA5_FOLDER}/era5_d2m_{year}.nc"
    variable:    d2m          # ERA5 dew-point temperature (K) — triggers d2m→q conversion
    sp_variable: sp
    sp_path:     "${ERA5_FOLDER}/era5_sp_{year}.nc"
    resample:    daily        # applied to q after the hourly conversion

  evspsbl:
    source: local
    path:         "${ERA5_FOLDER}/era5_e_{year}.nc"
    variable:     e
    scale_factor: -0.011574074074074073   # = -1000/86400: flips the sign (ERA5 e is
                              # negative for evaporation) and turns each hourly
                              # accumulated depth (m) into its share of a daily-mean
                              # kg m-2 s-1 rate
    resample: daily_sum       # SUM the hourly accumulations; 'daily' (mean) divides
                              # the daily total by 24 (the 3.6x evspsbl bug, 2026-10)

# ── Historical model (same CMIP6 model, historical experiment) ───────────────
# Option A — raw CMIP6 output via ESGF (OPeNDAP streaming; requires: pip install requests):
historical:
  tas:
    source: cmip6
    uri:    "cmip6://{model}/historical/tas/day/r1i1p1f1"

  pr:
    source: cmip6
    uri:    "cmip6://{model}/historical/pr/day/r1i1p1f1"

  # Both rsds/rsus and rlds/rlus are available daily for MPI-ESM1-2-HR historical.
  # ESGF query confirms 21 datasets for each.  See guides/radiation-bc.md.
  net_sw:
    composite:
      minuend:       # rsds — downwelling shortwave
        source: cmip6
        uri:    "cmip6://{model}/historical/rsds/day/r1i1p1f1"
      subtrahend:    # rsus — upwelling shortwave (reflected)
        source: cmip6
        uri:    "cmip6://{model}/historical/rsus/day/r1i1p1f1"

  net_lw:
    composite:
      minuend:       # rlds — downwelling longwave
        source: cmip6
        uri:    "cmip6://{model}/historical/rlds/day/r1i1p1f1"
      subtrahend:    # rlus — upwelling longwave (surface emission)
        source: cmip6
        uri:    "cmip6://{model}/historical/rlus/day/r1i1p1f1"

  uas:
    source: cmip6
    uri:    "cmip6://{model}/historical/uas/day/r1i1p1f1"

  vas:
    source: cmip6
    uri:    "cmip6://{model}/historical/vas/day/r1i1p1f1"

  ps:
    source: cmip6
    uri:    "cmip6://{model}/historical/ps/day/r1i1p1f1"

  psl:
    source: cmip6
    uri:    "cmip6://{model}/historical/psl/day/r1i1p1f1"

  huss:
    source: cmip6
    uri:    "cmip6://{model}/historical/huss/day/r1i1p1f1"

  # evspsbl: two options — use whichever frequency is available for your model.
  #
  # Option A-1: monthly evspsbl (Amon table).
  # Many CMIP6 models (e.g. MPI-ESM1-2-HR) only publish evspsbl at monthly
  # frequency.  QDM will correct the monthly distribution against ERA5 daily;
  # sub-daily disaggregation is still driven by the hourly ERA5 diurnal cycle.
  evspsbl:
    source: cmip6
    uri:    "cmip6://{model}/historical/evspsbl/Amon/r1i1p1f1"

  # Option A-2: daily latent-heat flux (hfls, day table) → convert to evspsbl.
  # hfls [W m⁻²] = evspsbl [kg m⁻² s⁻¹] × Lv [J kg⁻¹].
  # Divide by Lv ≈ 2.5e6 J kg⁻¹ to recover kg m⁻² s⁻¹.
  # Use this when daily output is needed but evspsbl/day is absent.
  # Uncomment and comment out Option A-1 above to switch.
  # evspsbl:
  #   source: cmip6
  #   uri:          "cmip6://{model}/historical/hfls/day/r1i1p1f1"
  #   scale_factor: 4.0e-7   # 1 / 2.5e6  (W m⁻² → kg m⁻² s⁻¹)

# Option B — ISIMIP3b bias-adjusted input data (requires: pip install isimip-client).
# Comment out Option A and uncomment below to use ISIMIP forcings directly.
# historical_isimip:

future_model:

  # Option A-2: daily latent-heat flux → evspsbl conversion.
  # Uncomment and comment out Option A-1 above to switch.
  # evspsbl:

  #   uri:          "cmip6://{model}/{scenario}/hfls/day/r1i1p1f1"

# ── Temporal disaggregation ───────────────────────────────────────────────────
# Elevates bias-corrected daily output to sub-daily frequency using the
# climatological diurnal cycle from ERA5.  Works with all three correction
# methods (delta / qm / qdm).
#
# Algorithm:
#   additive  (+): future_hourly = daily_bc + ERA5_diurnal_anomaly(DOY, hour)
#   multiplicative (*): future_hourly = daily_bc * ERA5_diurnal_ratio(DOY, hour)
#
# daily_bc daily-mean of future_hourly == daily_bc  (by construction)
#
# Point each variable to its ERA5 sub-daily file(s); use {year} placeholder
# for multi-year archives.  Only variables listed here are disaggregated.
#
disaggregation:
  enabled: true
  smooth_window: 15     # half-width (days) of circular DOY-smoothing window;
                        # set to 0 to disable smoothing
  variables:
    tas:
      era5_path:     "${ERA5_FOLDER}/era5_t2m_{year}.nc"
      era5_variable: t2m         # ERA5 t2m and corrected tas are both in Kelvin
    pr:
      era5_path:     "${ERA5_FOLDER}/era5_tp_{year}.nc"
      era5_variable: tp
      # ERA5 tp is in m per time step; CMIP6 pr is in kg m-2 s-1.
      # The disaggregation diurnal-cycle computation uses ratios, so the
      # unit mismatch cancels out — scale_factor is NOT needed here either.
    # ERA5 ssr and str provide the diurnal cycle of NET radiation directly.
    # ssr diurnal cycle: strong daytime peak, zero at night → multiplicative (*).
    # str diurnal cycle: small amplitude (dominated by strd, which is near-constant)
    #   → additive (+) consistent with net_lw kind.
    net_sw:
      era5_path:     "${ERA5_FOLDER}/era5_ssr_{year}.nc"
      era5_variable: ssr
      scale_factor: 0.000277778  # J/m² → W/m²  (ERA5 hourly accumulation ÷ 3600 s)
    net_lw:
      era5_path:     "${ERA5_FOLDER}/era5_str_{year}.nc"
      era5_variable: str
      scale_factor: 0.000277778  # J/m² → W/m²  (same)
    uas:
      era5_path:     "${ERA5_FOLDER}/era5_u10_{year}.nc"
      era5_variable: u10
    vas:
      era5_path:     "${ERA5_FOLDER}/era5_v10_{year}.nc"
      era5_variable: v10
    ps:
      era5_path:     "${ERA5_FOLDER}/era5_sp_{year}.nc"
      era5_variable: sp
      # ERA5 sp in Pa, CMIP6 ps in Pa — no scale_factor needed.
    psl:
      era5_path:     "${ERA5_FOLDER}/era5_mslp_{year}.nc"
      era5_variable: mslp
      # ERA5 mslp in Pa, CMIP6 psl in Pa — no scale_factor needed.
    huss:
      era5_path:     "${ERA5_FOLDER}/era5_d2m_{year}.nc"
      era5_variable: d2m          # ERA5 dew-point — triggers d2m→q conversion
      sp_variable:   sp
      sp_path:       "${ERA5_FOLDER}/era5_sp_{year}.nc"
      # The diurnal ratio is computed from sub-daily q = f(d2m, sp), so the
      # unit mismatch between d2m (K) and huss (kg/kg) is resolved internally.
      # No scale_factor needed.
    evspsbl:
      era5_path:     "${ERA5_FOLDER}/era5_e_{year}.nc"
      era5_variable: e
      # ERA5 e is negative for evaporation; the diurnal ratio ERA5_e_hour /
      # ERA5_e_daily_mean uses ratios of like-signed values so the sign cancels.
      # No scale_factor needed for disaggregation.

# ── Online data cache ─────────────────────────────────────────────────────────
# Copernicus Marine and CMIP6 datasets are saved here after the first download.
cache:
  enabled: true
  dir:     "${CACHE_ROOT}/cmip6"

# ── Output ────────────────────────────────────────────────────────────────────
# analyses_dir is used by bc-diagnostics to place outputs under
#   <analyses_dir>/scenarios/<scenario>/   (BCDiagnosticsLayout convention)
# dir is used by bc-correct for the bias-corrected NetCDF files.
# {scenario} in dir and filename patterns is substituted from metadata.scenario.
output:
  analyses_dir:     "${OCEANICU_ANALYSES_FOLDER}"
