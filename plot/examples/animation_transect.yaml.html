# ── animation_transect.yaml ───────────────────────────────────────────────────
# Vertical x-z (or y-z) transect animation.
#
# Usage:
#   ocean-animations --config examples/animation_transect.yaml
#   ocean-animations --config examples/animation_transect.yaml --show-grid
#
# The nearest single grid line to the given lat/lon is selected (no averaging).
# When z_var is set, actual sigma-level depths are computed from cumulative
# layer thicknesses so the curved sigma coordinate is shown correctly.
# ──────────────────────────────────────────────────────────────────────────────

plottype: transect

input:
  file: /data/kb/NS/Baseline/ns_monthly.nc
  variable: temp          # any 4-D scalar: temp, salt, …
  time_slice:
    start: 0
    end: 120              # 10 years of monthly output

transect:
  along: lon              # 'lon' → x-z section (horizontal axis = longitude)
                          # 'lat' → y-z section (horizontal axis = latitude)
  lat: 55.0               # latitude of the x-z section  (required for along: lon)
  # lon: 5.0              # longitude of the y-z section (required for along: lat)
  z_var: hn               # layer-height variable; omit to use coordinates attribute
  z_max: 200              # maximum depth [m]; omit to show full water column
  show_grid: false        # true → draw curving sigma-level interface lines
  # grid_color: white     # colour of grid lines (default: white)
  # grid_stride: 2        # draw every N-th grid line in x and z (default: 1)

output:
  file: temp_xz_55N.mp4
  fps: 4

title: "Temperature x-z transect at 55 °N"
colormap: thermal
vmin: 2.0
vmax: 20.0
