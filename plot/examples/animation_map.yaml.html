# ── animation_map.yaml ────────────────────────────────────────────────────────
# Single-panel horizontal map animation.
#
# Usage:
#   ocean-animations --config examples/animation_map.yaml
#
# plottype selects the animation kind — no --type flag needed.
# ──────────────────────────────────────────────────────────────────────────────

plottype: map

input:
  file: /data/ERA5/NA/era5_t2m_2019.nc
  variable: t2m
  time_slice:
    start: 0
    end: 365      # number of frames; omit for full file

# Spatial subset — omit to use the full domain
subset:
  lat_min: 48
  lat_max: 65
  lon_min: -12
  lon_max: 12

output:
  # file: t2m_daily_NS.mp4   # omit for auto-naming from variable long_name
  fps: 15

# title: "2 m Air Temperature — North Sea 2019"  # omit for auto-title from variable metadata
# colormap: thermal  # omit for auto-selection from variable metadata (thermal/balance/haline/…)
# vmin: -5.0         # omit for auto-range from data
# vmax: 25.0

# ── Optional time series panels ───────────────────────────────────────────────
# Uncomment to add panels showing the time series at up to 3 points.
# Each point gets a coloured dot on the map and its own subplot.
# Full time series is drawn as a thin gray background; the growing coloured
# line advances with the animation.
#
# ts_layout: below     # 'below' (default) or 'right'
timeseries:
  - label: "St. Malo"
    lat: 48.9
    lon: -1.8
    color: cyan
  - label: "Bristol Channel"
    lat: 51.54
    lon: -2.80
    color: yellow
  - label: "Wadden Sea"
    lat: 54.0
    lon:  8.5
    color: deepskyblue
