# ── plot_map.yaml ──────────────────────────────────────────────────────────────
# Static map plot from a NetCDF file.
#
# Usage:
#   ocean-plot --config examples/plot_map.yaml           # render and save
#   ocean-plot --config examples/plot_map.yaml --show    # open in viewer after saving
#   ocean-plot --config examples/plot_map.yaml --time-index 30
#   ocean-plot --config examples/plot_map.yaml --time-index mean
#   ocean-plot --info /data/model.nc                     # inspect variables first
# ──────────────────────────────────────────────────────────────────────────────

plottype: map          # 'map' (single frame) or 'map_mean' (time mean of all frames)

input:
  file: /data/ERA5/NA/era5_t2m_2019.nc
  variable: t2m
  # Optional: restrict which frames are included before the time selection.
  # time_slice:
  #   start: 0
  #   end: 90      # frame indices; omit 'end' to read to the last frame

# ── Time selection ────────────────────────────────────────────────────────────
# Integer → that specific frame index (0-based).
# 'mean'  → time-mean over all loaded frames (equivalent to plottype: map_mean).
# Omit    → first frame.
# Can also be supplied on the command line: --time-index 30 / --time-index mean
#
# time_index: 0

# ── Spatial subset ────────────────────────────────────────────────────────────
# Omit this section to use the full domain.
subset:
  lat_min: 48
  lat_max: 65
  lon_min: -12
  lon_max: 12

output:
  # file: t2m_plot.png    # omit for auto-naming: <variable>_plot.png
  dpi: 150                 # output file resolution

# title: "2 m Air Temperature — North Sea"   # omit to auto-derive from long_name
# colormap: thermal     # omit to auto-select (thermal/balance/haline/…)
# vmin: -5.0            # omit to use data range
# vmax: 25.0

# ── Annotations ───────────────────────────────────────────────────────────────
# Overlay static geographic annotations on the map.
