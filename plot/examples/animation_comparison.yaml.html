# ── animation_comparison.yaml ─────────────────────────────────────────────────
# Two-panel side-by-side map animation — e.g. model vs satellite observations.
# A horizontal colorbar sits between the maps and the optional time-series
# panels.  Use colorbar: shared (default) or individual per panel.
#
# Usage:
#   ocean-animations --config examples/animation_comparison.yaml
# ──────────────────────────────────────────────────────────────────────────────

plottype: comparison

# ── Left panel (typically model) ──────────────────────────────────────────────
left:
  input:
    file: /data/kb/NS/Baseline/ns_daily_surface.nc
    variable: temp
    time_slice:
      start: 0
      end: 365
  title: "pyGETM SST"
  # vmin:        # override shared vmin for this panel only
  # vmax:
  # colormap:    # per-panel colormap (overrides top-level colormap)
  # offset:      # additive transform applied after loading: value = value * factor + offset
  # factor:      # multiplicative transform (default 1.0)

# ── Right panel (typically observations) ──────────────────────────────────────
right:
  input:
    file: /data/CMEMS/sst/cmems_sst_daily_2019.nc
    variable: analysed_sst
    time_slice:
      start: 0
      end: 365
  title: "CMEMS SST (satellite)"
  # offset: -273.15   # example: convert K → °C for CCI/CMEMS SST products

# ── Spatial subset (applied to both panels) ───────────────────────────────────
subset:
  lat_min: 50
  lat_max: 62
  lon_min: -5
  lon_max: 10

output:
  file: sst_model_vs_obs_2019.mp4
  fps: 10

# title: "North Sea SST — Model vs Satellite 2019"
# Omit to auto-derive: "<left.title> / <right.title>"

# ── Colour scale ──────────────────────────────────────────────────────────────
colormap: thermal
vmin: 4.0
vmax: 22.0

# colorbar: shared       # 'shared' (default) — one colorbar for both panels
# colorbar: individual   # separate colorbar per panel; range set per panel
#                        # (use left.vmin/vmax and right.vmin/vmax, or auto)
#
# The colour range is always computed from the animated time window only,
# not the full dataset, so a time_slice doesn't skew the auto-range.

# ── Time synchronisation ──────────────────────────────────────────────────────
# 'index'    — advance both panels by the same frame index (default).
#              Requires equal-length time axes.
# 'calendar' — for each left frame, find the nearest right frame by date.
#              Use this when model and obs have different time resolutions
#              (e.g. daily model vs 4-day satellite composites).
time_sync: index

# ── Optional time series panels ───────────────────────────────────────────────
# Adds subplot rows below both maps.  Each panel shows BOTH datasets:
# a solid line for the left (model) and a dashed line for the right (obs).
# A thin background trace shows the full animated window; a growing coloured
# line advances with the animation.
#
# timeseries:
#   - label: "Central North Sea"
#     lat: 57.0
#     lon:  2.0
#     color: white
#     # obs:                    # optional in-situ overlay (buoy / station)
#     #   file: buoy_nns.csv    # CSV, TXT, or NetCDF
#     #   time_col: 0           # column index or name (CSV/TXT)
#     #   value_col: 1
#     #   # variable: temp      # NetCDF variable (NetCDF only)
#     #   label: "Buoy"
#     #   color: white
#     #   marker: "x"
#   - label: "English Channel"
#     lat: 51.0
#     lon:  1.5
#     color: yellow

# ── Optional annotations ──────────────────────────────────────────────────────
# Overlay static or time-gated geographic annotations on the map.
# Remove the leading '#' from any block to activate it.
#
# annotations:
#
#   # ── text label ────────────────────────────────────────────────────────────
#   - type: text
#     lat: 57.0
#     lon: 4.0
#     text: "North Sea"
#     fontsize: 11          # default 10
#     color: white          # default white
#     ha: center            # left/center/right, default left
#     va: bottom            # top/center/bottom, default bottom
#     fontweight: normal    # normal/bold, default normal
#     bbox:                 # optional background box (remove key to omit)
#       facecolor: black
#       alpha: 0.5
#       pad: 2
#     # time_on: 0          # frame index or ISO date string; omit = always on
#     # time_off: 100       # omit = stays on once activated
#
#   # ── marker (with optional text label) ────────────────────────────────────
#   - type: point
#     lat: 54.5
#     lon: 6.0
#     marker: "*"           # default "o"
#     markersize: 10        # default 8
#     color: red
#     label: "Station A"   # optional; creates a second text artist
#     # time_on: 30
#
#   # ── rectangle outline (optionally filled) ────────────────────────────────
#   - type: box
#     lat_min: 54.0
#     lat_max: 58.0
#     lon_min: 3.0
#     lon_max: 7.0
#     color: yellow
#     linewidth: 1.5        # default 1.0
#     alpha: 0.8            # default 1.0
#     fill: false           # true = semi-transparent fill, default false
#     label: ""             # optional text at top-left corner
#     # time_on: 50
#
#   # ── polyline through [lat, lon] waypoints ─────────────────────────────────
#   - type: line
#     points:
#       - [55.0, -2.0]
#       - [55.0,  8.0]
#     color: cyan
#     linewidth: 1.0
#     linestyle: "--"       # default "-"
#     label: "Section A"   # optional; placed at midpoint
#     # time_on: 0
#
#   # ── directed arrow ────────────────────────────────────────────────────────
#   - type: arrow
#     from_lat: 55.0
#     from_lon: 2.0
