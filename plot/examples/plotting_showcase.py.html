Demonstrates all plotting capabilities of the ocean plotting library
with matplotlib (OceanPlotter) and, separately, plain plotly figures.

Run this script to generate example plots showcasing:
1. Simple maps
2. Subplot grids (monthly, seasonal, comparison)
3. Time series
4. Scatter plots
5. Statistical distributions
6. Interactive plots (Plotly)
"""

def generate_example_data(nx=100, ny=100):
    """Generate realistic example ocean data."""

    lon = np.linspace(-10, 5, nx)
    lat = np.linspace(50, 60, ny)
    lon_2d, lat_2d = np.meshgrid(lon, lat)

    # Create realistic SST field with gradients
    sst_base = 12 + (lat_2d - 55) * 0.8 - (lon_2d + 2.5)**2 * 0.05

    # Add noise
    sst = sst_base + np.random.randn(ny, nx) * 0.3

    # Create monthly variation
    monthly_sst = []
    for month in range(12):
        seasonal = 5 * np.sin((month - 3) * np.pi / 6)
        monthly_sst.append(sst_base + seasonal + np.random.randn(ny, nx) * 0.5)

    # Create salinity
    salinity = 35 + (lat_2d - 55) * 0.05 + np.random.randn(ny, nx) * 0.1

    return {
        'lon': lon,
        'lat': lat,
        'sst': sst,
        'salinity': salinity,
        'monthly_sst': monthly_sst
    }

def run_showcase(engine='matplotlib'):
    """Run complete showcase."""

    print("\n" + "#"*60)
    print(f"# OCEAN PLOTTING LIBRARY SHOWCASE - {engine.upper()}")
    print("#"*60)

    # Create output directory
    output_dir = Path('showcase_outputs') / engine
    output_dir.mkdir(parents=True, exist_ok=True)

    # Initialize plotter
    plotter = create_plotter()  # the plotly showcases use plotly directly

    # Generate example data
    print("\nGenerating example data...")
    data = generate_example_data()

    # Run showcases based on engine
    if engine == 'matplotlib':
        showcase_1_simple_map(plotter, data, output_dir)
        showcase_2_monthly_grid(plotter, data, output_dir)
        showcase_3_monthly_alt(plotter, data, output_dir)
        showcase_4_comparison(plotter, data, output_dir)
        showcase_5_bias_analysis(plotter, data, output_dir)
        showcase_6_seasonal(plotter, data, output_dir)
        showcase_7_salinity(plotter, data, output_dir)
        showcase_8_custom_colormap(plotter, data, output_dir)
        showcase_9_tidal_analysis(plotter, data, output_dir)
        showcase_10_high_resolution(plotter, output_dir)

    elif engine == 'plotly':
        showcase_plotly_1_simple_map(plotter, data, output_dir)
        showcase_plotly_2_time_series(plotter, data, output_dir)
        showcase_plotly_3_scatter(plotter, data, output_dir)
        showcase_plotly_4_heatmap(plotter, data, output_dir)
        showcase_plotly_5_contour(plotter, data, output_dir)
        showcase_plotly_6_3d_surface(plotter, data, output_dir)
        showcase_plotly_7_animated(plotter, data, output_dir)
        showcase_plotly_8_subplots(plotter, data, output_dir)

    print("\n" + "="*60)
    print(f"SHOWCASE COMPLETE - {engine.upper()}")
    print(f"Output directory: {output_dir}")
    print("="*60)

    if engine == 'matplotlib':
        print(f"\nGenerated {len(list(output_dir.glob('*.png')))} PNG plots")
    else:
        print(f"\nGenerated {len(list(output_dir.glob('*.html')))} HTML plots")
        print("\nOpen any .html file in your browser to interact!")

def showcase_plotly_3_scatter(plotter, data, output_dir):
    """Plotly 3: Interactive scatter plot (Model vs Obs)."""

    print("\n" + "="*60)
    print("PLOTLY 3: INTERACTIVE SCATTER PLOT")
    print("="*60)

    try:
        import plotly.graph_objects as go

        # Flatten data for scatter
        model_flat = data['sst'].flatten()
        obs_flat = model_flat + np.random.randn(*model_flat.shape) * 0.8

        # Remove NaNs
        mask = ~(np.isnan(model_flat) | np.isnan(obs_flat))
        model_flat = model_flat[mask]
        obs_flat = obs_flat[mask]

        # Calculate statistics
        if HAS_SCIPY:
            slope, intercept, r_value, p_value, std_err = stats.linregress(obs_flat, model_flat)  # type: ignore[possibly-undefined]
        else:
            # Simple correlation
            r_value = np.corrcoef(obs_flat, model_flat)[0, 1]
            slope = 1.0
            intercept = 0.0

        # Create scatter
        fig = go.Figure()

        # Add scatter points
        fig.add_trace(go.Scatter(
            x=obs_flat,
            y=model_flat,
            mode='markers',
            name='Data',
            marker=dict(
                size=4,
                color=obs_flat,
                colorscale='thermal',
                opacity=0.6,
                colorbar=dict(title='Obs (°C)')
            )
        ))

        # Add 1:1 line
        plot_range = [min(obs_flat.min(), model_flat.min()),
                      max(obs_flat.max(), model_flat.max())]
        fig.add_trace(go.Scatter(
            x=plot_range,
            y=plot_range,
            mode='lines',
            name='1:1 Line',
            line=dict(color='black', dash='dash', width=2)
        ))

        # Add regression line
        reg_line = slope * np.array(plot_range) + intercept  # type: ignore[operator]
        fig.add_trace(go.Scatter(
            x=plot_range,
            y=reg_line,
            mode='lines',
            name=f'Fit (R²={r_value**2:.3f})',  # type: ignore[operator]
            line=dict(color='red', width=2)
        ))

        # Update layout
        fig.update_layout(
            title=f'Model vs Observations<br>R² = {r_value**2:.3f}, Bias = {np.mean(model_flat - obs_flat):.2f}°C',  # type: ignore[operator]
            xaxis_title='Observations (°C)',
            yaxis_title='Model (°C)',
            width=700,
            height=700,
            template='plotly_white'
        )

        # Equal aspect ratio
        fig.update_xaxes(scaleanchor="y", scaleratio=1)

        filename = output_dir / '03_scatter_validation.html'
        fig.write_html(filename)

        print(f"✓ Created: {filename}")
        print("  → Hover over points for values")

    except ImportError as e:
        print(f"✗ Skipping: {e}")

def showcase_plotly_7_animated(plotter, data, output_dir):
    """Plotly 7: Animated monthly evolution."""

    print("\n" + "="*60)
    print("PLOTLY 7: ANIMATED MAP")
    print("="*60)

    try:
        import plotly.graph_objects as go

        months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
                 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']

        # Create frames for animation
        frames = []
        for month_idx, month_data in enumerate(data['monthly_sst']):
            frames.append(go.Frame(
                data=[go.Heatmap(
                    x=data['lon'],
                    y=data['lat'],
                    z=month_data,
                    colorscale='thermal',
                    zmin=np.min([d.min() for d in data['monthly_sst']]),
                    zmax=np.max([d.max() for d in data['monthly_sst']]),
                    colorbar=dict(title='SST (°C)')
                )],
                name=months[month_idx]
            ))

        # Create figure with first frame
        fig = go.Figure(
            data=[go.Heatmap(
                x=data['lon'],
                y=data['lat'],
                z=data['monthly_sst'][0],
                colorscale='thermal',
                colorbar=dict(title='SST (°C)')
            )],
            frames=frames
        )

        # Add animation controls
        fig.update_layout(
            title='Monthly SST Evolution - Press Play!',
            xaxis_title='Longitude',
            yaxis_title='Latitude',
            width=900,
            height=700,
            updatemenus=[{
                'type': 'buttons',
                'showactive': False,

    """Plotly 8: Interactive subplots."""

    print("\n" + "="*60)
    print("PLOTLY 8: INTERACTIVE SUBPLOTS")
    print("="*60)

    try:
        from plotly.subplots import make_subplots
        import plotly.graph_objects as go

        # Create 2x2 subplot
        fig = make_subplots(
            rows=2, cols=2,
            subplot_titles=('Winter (DJF)', 'Spring (MAM)',
                          'Summer (JJA)', 'Autumn (SON)'),
            vertical_spacing=0.12,
            horizontal_spacing=0.1
        )

        # Calculate seasonal means
        seasons = {
            'Winter': [11, 0, 1],
            'Spring': [2, 3, 4],
            'Summer': [5, 6, 7],
            'Autumn': [8, 9, 10]
        }

        positions = [(1,1), (1,2), (2,1), (2,2)]

        for (row, col), (season, months) in zip(positions, seasons.items()):
            seasonal_mean = np.mean([data['monthly_sst'][m] for m in months], axis=0)

            fig.add_trace(
                go.Heatmap(
                    x=data['lon'],
                    y=data['lat'],
                    z=seasonal_mean,
                    colorscale='thermal',
                    showscale=(row==1 and col==2),  # Only show colorbar once
                    colorbar=dict(title='SST (°C)', x=1.15) if (row==1 and col==2) else None
                ),
                row=row, col=col
            )

        fig.update_layout(
            title_text='Seasonal SST - Hover Each Panel!',
            width=1000,
            height=900,
            template='plotly_white'
        )

        # Update all subplots to have equal aspect ratio
        for i in range(1, 5):
            fig.update_yaxes(scaleanchor=f"x{i}", scaleratio=1, row=(i-1)//2+1, col=(i-1)%2+1)

        filename = output_dir / '08_seasonal_subplots.html'
        fig.write_html(filename)

        print(f"✓ Created: {filename}")
        print("  → Hover over each season independently!")
